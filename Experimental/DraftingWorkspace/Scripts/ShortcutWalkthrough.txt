# ① pick modes 1 / 2 and Ctrl+A outside a text field (select every edge in mode 2, every face otherwise)
frames 2
expect faces == 11
expect steps == 8
key 2

key A Ctrl
expect selected-edges == 24
key 1
expect pick-mode == 1
key A Ctrl
expect selected-faces == 11
expect selected-edges == 0
key Escape
expect selected-faces == 0

# ② B opens a fillet; moving left past zero turns the signed size into a chamfer; viewport right-click cancels
#    (selection changes are undoable, so the undo depth stays 5 across the cancelled session)
key 2
move world 2.0 0.0 0.0
frames 2
click world 2.0 0.0 0.0
expect selected-edges == 1
expect undo == 5
key B
frames 2
expect bevel-active == 1
expect bevel-chamfer == 0
move viewport 0.40 0.70
frames 1
move viewport 0.40 0.70 offset -300 0
frames 2
expect bevel-chamfer == 1
expect bevel-pinned == 0
capture 19BevelSignChamfer
click viewport 0.40 0.70 offset -300 0 Right
frames 1
expect bevel-active == 0
expect faces == 11
expect steps == 8
expect undo == 5
